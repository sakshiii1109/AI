# Model-Based Intelligent Agent

grid = [
    ['S', '.', '#'],
    ['.', '#', '.'],
    ['.', '.', 'G']
]

position = [0, 0]

# Internal state
visited = []

while True:
    x, y = position
    visited.append((x, y))

    print("\nCurrent Position:", position)

    if grid[x][y] == 'G':
        print("Goal Reached!")
        break

    # Possible movements: Down, Right, Up, Left
    moves = [(1, 0), (0, 1), (-1, 0), (0, -1)]

    moved = False

    for dx, dy in moves:
        nx = x + dx
        ny = y + dy

        if (0 <= nx < 3 and 0 <= ny < 3
            and grid[nx][ny] != '#'
            and (nx, ny) not in visited):

            position = [nx, ny]
            moved = True
            break

    if not moved:
        print("Goal cannot be reached!")
        break

print("\nVisited Positions:", visited)

visited = set()

def dfs(a, b):

    if (a, b) in visited:
        return False

    visited.add((a, b))
    print("Jug 1:", a, "Jug 2:", b)

    # Check goal
    if a == target or b == target:
        print("Target Reached!")
        return True

    # Possible operations
    states = [
        (capacity1, b),  # Fill Jug 1
        (a, capacity2),  # Fill Jug 2
        (0, b),          # Empty Jug 1
        (a, 0),          # Empty Jug 2
        (a - min(a, capacity2-b), b + min(a, capacity2-b)),
        (a + min(b, capacity1-a), b - min(b, capacity1-a))
    ]

    for x, y in states:
        if dfs(x, y):
            return True

    return False

# Blocks World using BFS

from collections import deque

start = ("A", "B")   # B is on A
goal = ("B", "A")    # A is on B

queue = deque([[start]])
visited = set()

while queue:
    path = queue.popleft()
    state = path[-1]

    if state == goal:
        print("Goal Reached!")
        print("Steps:")
        for step in path:
            print(step)
        break

    if state not in visited:
        visited.add(state)

        # Move blocks
        next_states = [
            ("A", "B"),
            ("B", "A")
        ]

        for new_state in next_states:
            if new_state not in visited:
                queue.append(path + [new_state])

else:
    print("Goal not found")

# 8 Puzzle Problem using A* Search

start = [1, 2, 3,
         4, 0, 6,
         7, 5, 8]

goal = [1, 2, 3,
        4, 5, 6,
        7, 8, 0]

def heuristic(state):
    count = 0
    for i in range(9):
        if state[i] != 0 and state[i] != goal[i]:
            count += 1
    return count

def get_moves(state):
    moves = []
    zero = state.index(0)
    row = zero // 3
    col = zero % 3

    directions = [(-1, 0), (1, 0), (0, -1), (0, 1)]

    for dr, dc in directions:
        r = row + dr
        c = col + dc

        if 0 <= r < 3 and 0 <= c < 3:
            new = state[:]
            pos = r * 3 + c
            new[zero], new[pos] = new[pos], new[zero]
            moves.append(new)

    return moves

# A* Algorithm
open_list = [(heuristic(start), 0, start, [start])]
visited = set()

while open_list:
    open_list.sort(key=lambda x: x[0])
    f, g, state, path = open_list.pop(0)

    if state == goal:
        print("Goal Reached!")
        print("Steps:", len(path) - 1)

        for s in path:
            print(s[0:3])
            print(s[3:6])
            print(s[6:9])
            print()

        break

    key = tuple(state)

    if key in visited:
        continue

    visited.add(key)

    for new_state in get_moves(state):
        if tuple(new_state) not in visited:
            h = heuristic(new_state)
            open_list.append((g + 1 + h, g + 1,
                              new_state, path + [new_state]))
else:
    print("Solution not found")

# 8 Queens Problem using Backtracking

N = 8

board = [-1] * N

def is_safe(row, col):
    for i in range(row):
        if board[i] == col:
            return False

        if abs(board[i] - col) == abs(i - row):
            return False

    return True


def solve(row):
    if row == N:
        return True

    for col in range(N):
        if is_safe(row, col):
            board[row] = col

            if solve(row + 1):
                return True

            board[row] = -1

    return False


if solve(0):
    print("Solution Found!")

    for row in range(N):
        for col in range(N):
            if board[row] == col:
                print("Q", end=" ")
            else:
                print(".", end=" ")
        print()
else:
    print("No Solution")

# Sudoku Solver using Constraint Propagation

grid = [
    [5, 3, 0, 0, 7, 0, 0, 0, 0],
    [6, 0, 0, 1, 9, 5, 0, 0, 0],
    [0, 9, 8, 0, 0, 0, 0, 6, 0],
    [8, 0, 0, 0, 6, 0, 0, 0, 3],
    [4, 0, 0, 8, 0, 3, 0, 0, 1],
    [7, 0, 0, 0, 2, 0, 0, 0, 6],
    [0, 6, 0, 0, 0, 0, 2, 8, 0],
    [0, 0, 0, 4, 1, 9, 0, 0, 5],
    [0, 0, 0, 0, 8, 0, 0, 7, 9]
]

def possible(row, col, num):
    for i in range(9):
        if grid[row][i] == num:
            return False
        if grid[i][col] == num:
            return False

    r = (row // 3) * 3
    c = (col // 3) * 3

    for i in range(r, r + 3):
        for j in range(c, c + 3):
            if grid[i][j] == num:
                return False

    return True

def solve():
    for row in range(9):
        for col in range(9):
            if grid[row][col] == 0:
                for num in range(1, 10):
                    if possible(row, col, num):
                        grid[row][col] = num

                        if solve():
                            return True

                        grid[row][col] = 0

                return False

    return True

if solve():
    print("Solved Sudoku:")
    for row in grid:
        print(row)
else:
    print("No solution exists")

# Alpha-Beta Pruning Algorithm

def alphabeta(depth, node, maximizing, alpha, beta):

    # Leaf node values
    if depth == 2:
        return tree[node]

    if maximizing:
        best = -999

        for child in range(2):
            value = alphabeta(
                depth + 1, node * 2 + child,
                False, alpha, beta
            )

            best = max(best, value)
            alpha = max(alpha, best)

            if beta <= alpha:
                break

        return best

    else:
        best = 999

        for child in range(2):
            value = alphabeta(
                depth + 1, node * 2 + child,
                True, alpha, beta
            )

            best = min(best, value)
            beta = min(beta, best)

            if beta <= alpha:
                break

        return best


# Leaf node values
